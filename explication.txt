🔹 1. ENCODAGE :

import org.apache.spark.ml.feature.StringIndexer

val indexer1 = new StringIndexer().setInputCol("TransactionType").setOutputCol("TransactionTypeIndex")
val indexer2 = new StringIndexer().setInputCol("Channel").setOutputCol("ChannelIndex")
val indexer3 = new StringIndexer().setInputCol("DeviceType").setOutputCol("DeviceTypeIndex")

val df1 = indexer1.fit(df).transform(df)
val df2 = indexer2.fit(df1).transform(df1)
val df3 = indexer3.fit(df2).transform(df2)
🔹 2. FEATURES VECTOR:
import org.apache.spark.ml.feature.VectorAssembler

val assembler = new VectorAssembler()
  .setInputCols(Array(
    "TransactionAmount",
    "LoginAttempts",
    "AccountBalance",
    "TransactionTypeIndex",
    "ChannelIndex",
    "DeviceTypeIndex"
  ))
  .setOutputCol("features")

val finalData = assembler.transform(df3)
🔹 3. MODÈLE ML :
import org.apache.spark.ml.classification.LogisticRegression

val lr = new LogisticRegression()
  .setLabelCol("Fraud")
  .setFeaturesCol("features")

val model = lr.fit(finalData)
🔹 4. PRÉDICTIONS :
val predictions = model.transform(finalData)
predictions.select("Fraud","prediction").show(10)
🔹 5. ÉVALUATION :
import org.apache.spark.ml.evaluation.BinaryClassificationEvaluator

val evaluator = new BinaryClassificationEvaluator()
  .setLabelCol("Fraud")

val accuracy = evaluator.evaluate(predictions)

println("Score = " + accuracy)
